// ===== FILE: src/lib/money.ts =====
export function formatCurrency(cents: number, currency = 'USD'): string {
  return new Intl.NumberFormat('en-US', { style: 'currency', currency }).format(cents / 100);
}

// ===== FILE: src/features/invoices/invoicePdf.ts =====
import { formatCurrency } from '../../lib/money';

function centsToDisplay(cents: number): string {
  const dollars = (cents / 100).toFixed(2);
  return `$${dollars}`;
}

export function renderInvoiceTotal(totalCents: number): string {
  return centsToDisplay(totalCents);
}

export function renderInvoiceTax(taxCents: number): string {
  return formatCurrency(taxCents);
}
